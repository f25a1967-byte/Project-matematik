    <script>
        // Generate realistic crop dataset using the functions
        // f(x) = -0.005x^3 + 0.15x^2 + 0.5x + 2
        function getWeight(x) {
            return (-0.005 * Math.pow(x, 3)) + (0.15 * Math.pow(x, 2)) + (0.5 * x) + 2;
        }

        // f'(x) = -0.015x^2 + 0.3x + 0.5
        function getDerivative(x) {
            return (-0.015 * Math.pow(x, 2)) + (0.3 * x) + 0.5;
        }

        const labels = [];
        const weightData = [];
        
        // Simulating data points for 24 weeks of fruit bunch development
        for (let w = 0; w <= 24; w++) {
            labels.push(w);
            weightData.push(getWeight(w).toFixed(2));
        }

        // Initializing Chart.js
        const ctx = document.getElementById('growthChart').getContext('2d');
        const growthChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [{
                    label: 'Fresh Fruit Bunch Weight (kg)',
                    data: weightData,
                    borderColor: '#0f766e', // Teal-700
                    borderWidth: 3,
                    backgroundColor: 'rgba(20, 184, 166, 0.1)',
                    fill: true,
                    tension: 0.3, // smooth curving
                    pointBackgroundColor: '#0f766e',
                    pointHoverRadius: 8,
                    pointHoverBackgroundColor: '#f59e0b' // Amber-500 indicator on hover
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        position: 'top',
                        labels: { font: { weight: 'bold' } }
                    }
                },
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: 'Time (Weeks Post-Flowering)',
                            font: { weight: 'bold' }
                        }
                    },
                    y: {
                        title: {
                            display: true,
                            text: 'Weight per Bunch (f(x) in kg)',
                            font: { weight: 'bold' }
                        },
                        min: 0,
                        max: 25
                    }
                },
                // Trigger action when the user clicks anywhere on the canvas
                onClick: (e) => {
                    const points = growthChart.getElementsAtEventForMode(e, 'nearest', { intersect: false }, true);
                    if (points.length) {
                        const firstPoint = points[0];
                        const week = growthChart.data.labels[firstPoint.index];
                        updateCalculusDisplay(week);
                    }
                }
            }
        });

        // Function to compute derivative and update HTML DOM elements dynamically
        function updateCalculusDisplay(week) {
            const currentWeight = getWeight(week).toFixed(2);
            const rateOfChange = getDerivative(week).toFixed(3);
            
            // Generate layman insights dynamically based on the value of the derivative
            let insight = "";
            if (week <= 4) {
                insight = `The bunch is in its infant stage. At ${rateOfChange} kg/week, the change is slow because cellular division has just started. Hold off harvesting.`;
            } else if (week > 4 && week <= 8) {
                insight = `Growth speed is accelerating cleanly. The derivative is healthy (${rateOfChange} kg/week), meaning nutrients from the soil are transitioning rapidly into fruit meat.`;
            } else if (week > 8 && week <= 12) {
                insight = `CRITICAL BIOMASS ACCUMULATION! The derivative peaks around week 10 at ~2.00 kg/week. This point of inflection means the plant is working at maximum biological efficiency.`;
            } else if (week > 12 && week <= 18) {
                insight = `The derivative is dropping (${rateOfChange} kg/week), though total weight still increases. The plant is hitting biological saturation. Plan harvesting logs immediately.`;
            } else {
                insight = `The derivative is approaching zero or turning negative (${rateOfChange} kg/week). The fruit is overripe and falling off the tree, ruining oil quality. Harvesting now hurts food security metrics.`;
            }

            // Manipulate DOM elements to show results
            document.getElementById('promptMessage').classList.add('hidden');
            document.getElementById('calcResult').classList.remove('hidden');
            
            document.getElementById('resWeek').innerText = week;
            document.getElementById('resWeight').innerText = currentWeight;
            document.getElementById('resRate').innerText = `${rateOfChange} kg/week`;
            document.getElementById('resInsight').innerText = insight;
        }
    </script>
